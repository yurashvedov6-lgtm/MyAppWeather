// 1. КЛАСС ЗАПУСКА ПРИЛОЖЕНИЯ (Связывает код с эмулятором)
class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContent {
            MaterialTheme {
                Surface(
                    modifier = Modifier.fillMaxSize(),
                    color = MaterialTheme.colorScheme.background
                ) {
                    // Здесь мы даем команду эмулятору показать ваш экран
                    MainScreen()
                }
            }
        }
    }
}

// 2. ВАШ РАБОЧИЙ ЭКРАН (Оставили без изменений, добавили только логику цифровой клавиатуры)
@Composable
fun MainScreen() {
    val inputTextState = rememberSaveable { mutableStateOf("") }
    val resultTextState = rememberSaveable { mutableStateOf("Результат: ") }

    Column(modifier = Modifier.padding(16.dp)) {
        OutlinedTextField(
            value = inputTextState.value,
            onValueChange = { inputTextState.value = it },
            label = { Text("Введите число") },
            shape = RoundedCornerShape(12.dp)
        )

        Spacer(modifier = Modifier.height(8.dp))
        Text(text = resultTextState.value)
        Spacer(modifier = Modifier.height(8.dp))

        Button(
            onClick = {
                val number = inputTextState.value.toIntOrNull()

                if (number != null) {
                    val square = number * number
                    resultTextState.value = "Результат: $square"
                } else {
                    resultTextState.value = "Ошибка: введите корректное число!"
                }
            },
            shape = RoundedCornerShape(12.dp)
        ) {
            Text("Возвести в квадрат")
        }
    }
}
